cmake_minimum_required(VERSION 3.8)
project(arm_hardware)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(arm_hardware SHARED src/arm_system.cpp)
ament_target_dependencies(arm_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface arm_hardware.xml)
install(TARGETS arm_hardware LIBRARY DESTINATION lib)
ament_export_libraries(arm_hardware)
ament_export_dependencies(hardware_interface pluginlib rclcpp)
if(BUILD_TESTING)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(serial_retry_test test/serial_retry.cpp)
  ament_target_dependencies(serial_retry_test hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_link_libraries(serial_retry_test Threads::Threads util)
  add_test(NAME serial_retry COMMAND serial_retry_test)
  add_executable(cad_mapping_test test/cad_mapping.cpp)
  ament_target_dependencies(cad_mapping_test hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_link_libraries(cad_mapping_test Threads::Threads util)
  add_test(NAME cad_mapping COMMAND cad_mapping_test)
endif()
ament_package()
